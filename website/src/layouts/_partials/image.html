{{- /* A raster image from assets/images as responsive WebP. Params: src, alt, width (largest), sizes, class, eager. An eager image is the page's largest above-the-fold image, so it also gets fetchpriority=high. */ -}} {{- $image := resources.Get (printf "images/%s" .src) }} {{- $max := .width | default 1200 }} {{- $widths := slice }} {{- range slice 480 800 $max }}{{ if and (le . $max) (le . $image.Width) }}{{ $widths = $widths | append . }}{{ end }}{{ end }} {{- $widths = $widths | uniq }} {{- $srcset := slice }} {{- $largest := "" }} {{- range $widths }} {{- $largest = $image.Resize (printf "%dx webp q82" .) }} {{- $srcset = $srcset | append (printf "%s %dw" $largest.RelPermalink .) }} {{- end }}
